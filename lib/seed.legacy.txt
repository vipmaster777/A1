export async function ensureSeedRoles() {
  const count = await prisma.roleProfile.count();
  if (count > 0) return;

  await prisma.roleProfile.createMany({
    data: seedRoles.map((role) => ({
      ...role,
      preferredZodiacSigns: JSON.stringify(role.preferredZodiacSigns),
      preferredBirthMonths: JSON.stringify(role.preferredBirthMonths),
    })),
  });
}
